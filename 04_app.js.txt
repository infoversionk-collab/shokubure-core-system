async function api(action,payload={}) {
  const user=auth.currentUser;if(!user)throw new Error("ログインしてください。");
  const idToken=await user.getIdToken();
  const res=await fetch(CORE_API_URL,{method:"POST",headers:{"Content-Type":"text/plain;charset=utf-8"},body:JSON.stringify({action,idToken,...payload})});
  if(!res.ok)throw new Error("CORE API通信エラー: "+res.status);
  const data=await res.json();if(!data.ok)throw new Error(data.message||data.error||"APIエラー");
  return data;
}
